def make_grid(n_d):
    x = np.linspace(-4.0, 4.0, n_d + 1)
    return x, x[1] - x[0]


def initial_psi(x):
    return ((alpha / np.pi) ** 0.25 * np.exp(-alpha / 2 * (x - x_start) ** 2)).astype(np.complex128)


def propagator(x, t):
    xb = x[:, None]
    xa = x[None, :]
    return np.sqrt(1 / (2j * np.pi * np.sin(t))) * np.exp(
        1j * ((xa**2 + xb**2) * np.cos(t) - 2 * xa * xb) / (2 * np.sin(t)))


def evolve(psi0, K, dx, n_steps):
    psi = np.empty((n_steps + 1, psi0.size), dtype=np.complex128)
    psi[0] = psi0
    for n in range(n_steps):
        psi[n + 1] = dx * K @ psi[n]
    return psi


x, dx = make_grid(N_D)
t = np.arange(N_steps + 1) * eps
print("x_start =", x_start)

\section*{Problem F}
\textbf{Prediction.}
As a function of $x_a$, the phase of $K_\epsilon(x_b, x_a)$ changes at the rate
\[
  \left|\frac{\partial \phi}{\partial x_a}\right| = \frac{|x_a \cos\epsilon - x_b|}{\sin\epsilon} \lesssim \frac{8}{\sin\epsilon}.
\]
The method requires $\Delta x$ to resolve this oscillation.
With $8\Delta x/\sin\epsilon$ as a measure of the phase change per grid step, the three cases give $0.28$ for (i), $8.7$ for (ii), and $8.7$ for (iii).
I therefore predict that (ii) and (iii) fail and (i) works.

\begin{figure}[H]
  \centering
  \includegraphics[width=0.65\textwidth]{figs/norm.pdf}
  \caption{$\mathcal{N}(t) = \Delta x \sum_i |\Psi_t(x_i)|^2$ without renormalization.}
\end{figure}

\noindent
Case (i) keeps $\mathcal{N}(T_0) = 0.9995$.
Case (ii) grows to $\mathcal{N}(T_0) = 7.4\times10^{257}$, and case (iii) to $2.6\times10^{51}$.

For small $\epsilon$, $K_\epsilon$ oscillates faster in $x_a$, and in the limit $\epsilon \to 0$ it approaches a $\delta$-function.
When the phase change per grid step becomes too large, the sum over the grid no longer approximates the integral.
$\Delta x\,K_\epsilon$ is then no longer unitary, and the norm grows exponentially with every step.

By Poisson summation, the grid sum fails once $|\partial\phi/\partial x_a|$ reaches $2\pi/\Delta x$ somewhere on the grid.
The condition for the method to work is $8\Delta x/\sin\epsilon < 2\pi$, i.e.
\[
  \epsilon_{\min} \approx \frac{4\Delta x}{\pi} = \frac{32}{\pi N_D}.
\]
For $N_D = 600$ this gives $\epsilon_{\min} \approx 0.017 \approx T_0/370$; numerically, the first failure is at $\epsilon = T_0/372$.
Doubling $N_D$ halves $\epsilon_{\min}$: the estimate is $T_0/740$ for $N_D = 1200$, and the first numerical failure is at $T_0/744$.

\begin{lstlisting}[language=Python]
def norm_vs_t(eps_c, n_d):
    xc, dxc = make_grid(n_d)
    n_steps = int(round(T0 / eps_c))
    psi_c = evolve(initial_psi(xc), propagator(xc, eps_c), dxc, n_steps)
    return np.arange(n_steps + 1) * eps_c, (np.abs(psi_c) ** 2).sum(axis=1) * dxc


plt.figure(figsize=(6, 4))
for n, n_d in ((16, 600), (512, 600), (128, 150)):
    t_c, N_c = norm_vs_t(T0 / n, n_d)
    plt.semilogy(t_c, N_c, label=rf"$\epsilon = T_0/{n}$, $N_D = {n_d}$")
    print(f"eps = T0/{n}, N_D = {n_d}: N(T0) = {N_c[-1]:.3e}")
plt.xlabel("time $t$")
plt.ylabel(r"$\mathcal{N}(t)$")
plt.legend()
plt.tight_layout()
plt.savefig("figs/norm.pdf")
plt.show()

for n_d in (600, 1200):
